"""Render actual aligned SharpaDex files, with no synthetic tactile values."""
from pathlib import Path
import json, subprocess
import cv2, numpy as np, pyarrow.parquet as pq
from PIL import Image, ImageDraw, ImageFont
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

R=Path(__file__).resolve().parent; O=R/'visualization';O.mkdir(exist_ok=True)
table=pq.read_table(R/'sample/data/chunk-000/episode_000000.parquet')
t=np.asarray(table['timestamp']);tac=np.asarray(table['observation.tactile'].to_pylist())
ann=json.loads((R/'sample/meta/annotations.jsonl').read_text().splitlines()[0])
keys=['head_left','head_right','wrist_left','wrist_right','tactile_raw','tactile_deform']
caps={k:cv2.VideoCapture(str(R/f'sample/videos/chunk-000/observation.images.{k}/episode_000000.mp4')) for k in keys}
fontpath='/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf'
font=lambda n:ImageFont.truetype(fontpath,n)
f22,f18,f15=font(22),font(18),font(15)
bg='#101920';fg='#e9f0f5';sub='#acbdca';colors=['#60a5fa','#f59e0b','#34d399','#f472b6','#c4b5fd']
fig,axes=plt.subplots(1,2,figsize=(16,2.55),dpi=100,facecolor=bg)
for hand,ax in enumerate(axes):
    ax.set_facecolor(bg)
    for i,name in enumerate(['thumb','index','middle','ring','little']):
        ax.plot(t,tac[:,hand*30+i*6+2],label=name,color=colors[i],linewidth=1)
    ax.set_title(('Left' if hand==0 else 'Right')+' fingertips: fz, stored values (unit not verified)',color=fg,fontsize=11)
    ax.set_xlim(0,len(t)/30);ax.set_xlabel('Seconds',color=sub,fontsize=10);ax.tick_params(colors=sub,labelsize=9)
    ax.grid(alpha=.15);ax.legend(ncol=5,fontsize=8,frameon=False,labelcolor=fg,loc='upper left')
    for s in ax.spines.values():s.set_color('#455563')
fig.subplots_adjust(left=.045,right=.985,bottom=.19,top=.86,wspace=.13);fig.canvas.draw()
plot=Image.fromarray(np.asarray(fig.canvas.buffer_rgba()).copy()).convert('RGB')
lims=[]
for ax in axes:
    x0,y0=ax.transData.transform((0,ax.get_ylim()[0]));x1,y1=ax.transData.transform((len(t)/30,ax.get_ylim()[1]));lims.append((x0,x1,255-y1,255-y0))
plot.save(O/'tactile_fz_timeseries.png');plt.close(fig)
W,H=1600,1120
cmd=['ffmpeg','-hide_banner','-loglevel','error','-y','-f','rawvideo','-pix_fmt','rgb24','-s',f'{W}x{H}','-r','10','-i','-','-an','-c:v','libx264','-preset','fast','-crf','23','-pix_fmt','yuv420p','-movflags','+faststart',str(O/'preview.mp4')]
p=subprocess.Popen(cmd,stdin=subprocess.PIPE)
def render(i,frames):
    img=Image.new('RGB',(W,H),bg);d=ImageDraw.Draw(img)
    d.text((20,12),'SharpaDex v1.0 | clean_plate_with_eraser | episode 0',font=f22,fill=fg)
    d.text((20,45),f't = {t[i]:.2f} s / {len(t)/30:.2f} s | source frame {i} | 30 Hz source, 10 fps preview at real-time speed',font=f15,fill=sub)
    for j,k in enumerate(keys[:4]):
        x=20+j*395;d.text((x,77),k.replace('_',' '),font=f18,fill=fg)
        img.paste(Image.fromarray(cv2.cvtColor(frames[k],cv2.COLOR_BGR2RGB)).resize((375,375)),(x,105))
    d.text((20,491),'Raw tactile: original 2 x 5 camera mosaic',font=f18,fill=fg)
    d.text((1000,491),'Deformation visualization: original layout',font=f18,fill=fg)
    img.paste(Image.fromarray(cv2.cvtColor(frames['tactile_raw'],cv2.COLOR_BGR2RGB)).resize((960,288)),(20,520))
    img.paste(Image.fromarray(cv2.cvtColor(frames['tactile_deform'],cv2.COLOR_BGR2RGB)).resize((600,240)),(1000,520))
    d.text((1000,773),'Finger-to-tile semantics not independently verified.',font=f15,fill=sub)
    phase=next((s['text'] for s in ann['language_segments'] if s['start_step']<=i<=s['end_step']),'Global task context; no local segment')
    d.text((20,824),'Annotation: '+phase,font=f15,fill=fg)
    img.paste(plot,(0,850));d=ImageDraw.Draw(img)
    for x0,x1,y0,y1 in lims:
        x=x0+(x1-x0)*i/len(t);d.line((x,850+y0,x,850+y1),fill='#ffffff',width=2)
    d.text((20,1100),'All panels use the same exported frame index. This verifies export alignment, not original hardware latency.',font=f15,fill=sub)
    return img
try:
    written=0
    for i in range(len(t)):
        frames={}
        for k,c in caps.items():
            ok,im=c.read();assert ok,(i,k);frames[k]=im
        if i%3:continue
        img=render(i,frames)
        if i==849:img.save(O/'poster.jpg',quality=92)
        p.stdin.write(img.tobytes());written+=1
    p.stdin.close();assert p.wait()==0
finally:
    for c in caps.values():c.release()
(O/'render.json').write_text(json.dumps(dict(source_frames=len(t),source_fps=30,source_seconds=len(t)/30,preview_frames=written,preview_fps=10,preview_seconds=written/10,poster_source_frame=849,alignment='same exported frame_index; original sensor latency not independently established'),indent=2))
print('Rendered',written,'frames')
